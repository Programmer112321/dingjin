height_metrics <- function(z) {

def height_metrics(h):
    """Height above ground array → dict of metrics. h should be pre-filtered (h > 2)."""
    if len(h) < 5:
        return None
    percs = np.percentile(h, [10, 25, 50, 75, 90, 95, 99])
    return {
        "H10":   percs[0], "H25": percs[1], "H50": percs[2],
        "H75":   percs[3], "H90": percs[4], "H95": percs[5], "H99": percs[6],
        "Hmean": h.mean(),
        "Hmax":  h.max(),
        "Hsd":   h.std(),
        "Hcv":   h.std() / h.mean() if h.mean() > 0 else np.nan,
        "Hskew": stats.skew(h),
        "Hkurt": stats.kurtosis(h),
        "Hqm":   np.sqrt(np.mean(h**2))
    }

# ── Pixel-level (20 m grid) ────────────────────────────────────────
def pixel_metrics(x, y, H, func, res=20):
    """Apply func over a regular res × res grid; returns dict of 2D arrays."""
    h_filt = H > 2
    xmin, xmax = x.min(), x.max()
    ymin, ymax = y.min(), y.max()
    cols = max(1, int(np.ceil((xmax - xmin) / res)))
    rows = max(1, int(np.ceil((ymax - ymin) / res)))
    ci = np.clip(((x - xmin) / res).astype(int), 0, cols - 1)
    ri = np.clip(((ymax - y) / res).astype(int), 0, rows - 1)
    out = {}
    for r in range(rows):
        for c in range(cols):
            mask = (ri == r) & (ci == c) & h_filt
            if mask.sum() < 5:
                continue
            m = func(H[mask])
            if m is None:
                continue
            for k, v in m.items():
                if k not in out:
                    out[k] = np.full((rows, cols), np.nan)
                out[k][r, c] = v
    return out

grid = pixel_metrics(x, y, H, height_metrics, res=20)
Metric Formula Ecological meaning
H50 50th percentile Median canopy height — robust central tendency
H95 / H99 95th / 99th percentile Dominant / emergent height; proxy for top-of-canopy
Hmean \(\bar{H}\) Mean canopy height — strongly correlated with AGB
Hsd \(\sigma_H\) Vertical spread — indicator of structural heterogeneity
Hcv \(\sigma_H / \bar{H}\) Scale-free variability — useful across biomes
Hskew \(\mu_3 / \sigma^3\) Positive skew = bottom-heavy profile (young/regrowth)
Hqm \(\sqrt{\bar{H^2}}\) Quadratic mean — more sensitive to tall trees than Hmean
Tip

H99 vs Hmax: Hmax is the single tallest point and is highly susceptible to surviving noise. H99 is far more robust and is what most published AGB models use.

def cover_metrics(h, ret, threshold=2.0):
    n_total  = len(h)
    n_above  = np.sum(h >= threshold)
    first    = ret == 1
    return {
        "CC":       n_above / n_total if n_total > 0 else np.nan,
        "GF":       1 - n_above / n_total if n_total > 0 else np.nan,
        "CC_first": np.mean(h[first] >= threshold) if first.sum() > 0 else np.nan
    }

# Grid version — uses the same pixel_metrics() helper from above
def cover_func(h):
    return cover_metrics(h, ret[H > -999], threshold=2.0)

Canopy cover (CC) from LiDAR:

\[CC = \frac{n_{\text{returns} \geq h_t}}{n_{\text{total returns}}} \times 100\%\]

Gap fraction (GF = 1 − CC) as a proxy for LAI via Beer-Lambert:

\[\text{LAI} \approx -\frac{\ln(\text{GF})}{k}, \quad k \approx 0.5\]

def density_profile(h):
    h   = h[h >= 0]
    if len(h) == 0:
        return None
    breaks = [0, 0.5, 2, 5, 10, 20, 30, np.inf]
    labels = ["G", "S0.5", "S2", "S5", "S10", "S20", "S30"]

    return {f"D_{lbl}": c / total for lbl, c in zip(labels, counts)}

Part 3 — Structural Complexity Metrics

3.1 Rumple Index (RI)

\[RI = \frac{A_{3D}}{A_{2D}}\]

\(A_{3D}\) — triangulated canopy surface area (TIN from first returns); \(A_{2D}\) — projected footprint. A flat uniform canopy has \(RI \approx 1\); a complex multi-layered canopy has \(RI \gg 1\).

def rumple_index_py(x, y, z, res=1.0):
    """Compute RI via Delaunay triangulation of first-return canopy surface."""
    # Keep only first returns (or highest-z per cell as a proxy)
    xi  = np.floor(x / res).astype(int)
    yi  = np.floor(y / res).astype(int)
    key = xi * 1_000_000_007 + yi
    _, top = np.unique(key, return_index=True)   # one pt per cell = highest
    px, py, pz = x[top], y[top], z[top]

    if len(px) < 4:
        return np.nan

    pts2d = np.column_stack([px, py])
    tri   = Delaunay(pts2d)

    # 3D surface area via cross product for each triangle
    verts = np.column_stack([px, py, pz])
    a3d   = 0.0
    for s in tri.simplices:
        v  = verts[s]
        ab = v[1] - v[0]
        ac = v[2] - v[0]
        a3d += np.linalg.norm(np.cross(ab, ac)) / 2

    a2d = (px.max() - px.min()) * (py.max() - py.min())
    return a3d / a2d if a2d > 0 else np.nan

ri = rumple_index_py(x, y, H, res=1.0)
RI value Canopy type
1.0–1.1 Flat agricultural / grassland
1.1–1.3 Young even-aged plantation
1.3–1.6 Mature mixed forest
> 1.6 Old-growth, multi-layered, high structural complexity
Important

RI is resolution-sensitive: a 0.25 m CHM captures branch-level rugosity; a 5 m CHM smooths it. Always report CHM resolution alongside RI.

3.3 Vertical Complexity Index (VCI) — Shannon Entropy

\[VCI = -\sum_{i=1}^{n} p_i \ln(p_i)\]

Normalised VCI = VCI / ln(n) ranges 0–1 (0 = single layer; 1 = uniform vertical distribution).

  • lidR
  • lasR
  • Python
vci_metric <- function(z, n_strata = 10) {
  z     <- z[z >= 0 & z < 80]
  strat <- cut(z, breaks = seq(0, 80, length.out = n_strata + 1),
               include.lowest = TRUE)
  p     <- prop.table(table(strat))
  p     <- p[p > 0]
  vci   <- -sum(p * log(p))
  list(VCI = vci, VCI_norm = vci / log(n_strata))
}

def vci_metric(h, n_strata=10):
    h = h[(h >= 0) & (h < 80)]
    if len(h) < 5:
        return {"VCI": np.nan, "VCI_norm": np.nan}
    edges  = np.linspace(0, 80, n_strata + 1)
    counts = np.histogram(h, bins=edges)[0]
    p      = counts / counts.sum()
    p      = p[p > 0]
    vci    = -np.sum(p * np.log(p))
    return {"VCI": vci, "VCI_norm": vci / np.log(n_strata)}

bimodality_coefficient <- function(x) {
  n <- length(x)
  (moments::skewness(x)^2 + 1) /
  (moments::kurtosis(x) + 3 * (n - 1)^2 / ((n - 2) * (n - 3)))
}

def bimodality_coefficient(h):
    n = len(h)
    if n < 4:
        return np.nan
    sk = stats.skew(h)
    ku = stats.kurtosis(h)   # Fisher (excess) kurtosis
    return (sk**2 + 1) / (ku + 3 * (n - 1)**2 / max((n - 2) * (n - 3), 1))

def shape_metrics(h):
    h = h[h >= 0]
    if len(h) < 4:
        return None
    return {
        "Hskew":  stats.skew(h),
        "Hkurt":  stats.kurtosis(h),
        "Hbimod": bimodality_coefficient(h),
        "Hrange": h.max() - h.min()
    }

3.6 Advanced Complexity: Adaptive KDE Entropy (Liu et al. 2022)

Simple height-bin VCI is sensitive to point density gradients that often exist across a tile due to scan-angle variation. The Liu et al. (2022) approach removes this bias before computing entropy:

  1. avg_point_dist — compute mean nearest-neighbour distance per 1 m height layer to quantify the vertical density gradient.
  2. adaptative_resample — iteratively coarsen voxel size until a Mann-Kendall trend test finds no significant vertical density trend (p > α), then return the resampled cloud.
  3. canopy_entropy — project the resampled cloud onto three orthogonal planes (XY, XZ, YZ), fit a KDE on each, compute Shannon entropy \(H = -\sum p \ln(p) \cdot dA\), and return \(H_{XYZ} = \sqrt{H_{XY}^2 + H_{XZ}^2 + H_{YZ}^2}\).
  • lidR
  • lasR
  • Python
library(lidR)
library(data.table)
library(nabor)      # fast kNN
library(TreeLS)     # las2xyz helper
library(ks)         # kernel density estimation
library(trend)      # Mann-Kendall test

# ── 1. Average nearest-neighbour distance per vertical layer ───────
avg_point_dist <- function(las, thickness = 1) {
  n_pts <- nrow(las@data)
  if (n_pts < 2) {
    return(data.table(layer = 0L, dist = NA_real_))
  }
  las@data$MinDist <- nabor::knn(TreeLS:::las2xyz(las), k = 2)$nn.dist[, 2]
  z_pts            <- las$Z - min(las$Z)
  las@data$layer   <- as.integer(z_pts / thickness)
  las@data[order(layer), .(dist = mean(MinDist, na.rm = TRUE)), by = layer]
}

# ── 2. Adaptive voxel resampling ───────────────────────────────────
adaptative_resample <- function(las, thickness = 1, alpha = 0.05, step = 0.1) {
  avg_dst <- avg_point_dist(las, thickness)

  noise_tol <- mean(avg_dst$dist, na.rm = TRUE) + 2 * sd(avg_dst$dist, na.rm = TRUE)
  avg_dst   <- avg_dst[!is.na(dist) & dist < noise_tol]
  if (nrow(avg_dst) < 3) return(NULL)

  max_dst <- max(avg_dst$dist)

  vlas <- las
  mkt  <- NULL

  for (ratio in seq(2, 5, step)) {
    vox     <- ratio * max_dst
    vlas    <- TreeLS::tlsSample(las, smp.voxelize(vox))
    avg_dst <- avg_point_dist(vlas, thickness)
    avg_dst <- avg_dst[!is.na(dist)]
    if (nrow(avg_dst) < 3) break
    mkt <- trend::mk.test(avg_dst$dist)
    if (mkt$p.value > alpha) break
  }

  if (is.null(mkt)) return(NULL)
  list(las = vlas, ratio = ratio, vox = vox, dist = avg_dst, mk.test = mkt)
}

# ── 3. KDE Shannon entropy in XY, XZ, YZ planes ───────────────────
canopy_entropy <- function(las, bw = 0.2, grid_size = 0.1) {
  bounds <- apply(las@data[, .(X, Y, Z)], 2, range)
  ce     <- numeric(3L)
  names(ce) <- c("XY", "XZ", "YZ")

  for (ij in list(c("X", "Y"), c("X", "Z"), c("Y", "Z"))) {
    i <- ij[1]; j <- ij[2]
    key   <- paste(ij, collapse = "")
    plane <- las@data[, ..ij]
    ni   <- 2L + as.integer((diff(bounds[, i]) + 8 * bw) %/% grid_size)
    nj   <- 2L + as.integer((diff(bounds[, j]) + 8 * bw) %/% grid_size)
    lims <- c(bounds[, ij]) + (bw * 4 + grid_size / 2) * c(-1, 1, -1, 1)
    den  <- ks::kde(plane, h = bw,
                    gridsize = c(ni, nj),
                    xmin     = lims[c(1, 3)],
                    xmax     = lims[c(2, 4)])$estimate
    den        <- den[den > 0]
    ce[key]    <- -sum(den * log(den) * grid_size^2)
  }

  ce["XYZ"] <- sqrt(sum(ce^2))
  as.data.frame(t(ce))
}

# ── 4. Entropy wrapper with adaptive resampling ────────────────────
get_entropy <- function(las, thickness = 1) {
  las@data$Z <- las@data$Height   # use HAG geometry (Liu et al. 2022)
  res <- adaptative_resample(las, thickness)
  if (is.null(res)) return(NULL)
  ent       <- canopy_entropy(res$las)
  ent$p     <- res$mk.test$p.value
  ent$vox   <- res$vox
  ent
}

pix_complexity <- function(x, y, z, h) {
  keep <- h >= 1

import numpy as np
from scipy import stats
from scipy.spatial import cKDTree
from scipy.stats import gaussian_kde

# ── 1. Average nearest-neighbour distance per vertical layer ───────
def avg_point_dist(x, y, z, thickness=1.0):
    """Mean NN-distance (k=2) per height layer of `thickness` m."""
    pts  = np.column_stack([x, y, z])
    tree = cKDTree(pts)
    d, _ = tree.query(pts, k=2)
    nn   = d[:, 1]              # distance to nearest neighbour (exclude self)

    z0     = z - z.min()
    layers = (z0 / thickness).astype(int)
    result = {}
    for lyr in np.unique(layers):
        m = layers == lyr
        result[lyr] = nn[m].mean()
    return result   # {layer_id: mean_dist}

# ── 2. Adaptive voxel resampling ───────────────────────────────────
def adaptative_resample(x, y, z, thickness=1.0, alpha=0.05, step=0.1):
    """Coarsen voxel size until Mann-Kendall finds no vertical density trend."""
    ld = avg_point_dist(x, y, z, thickness)
    if len(ld) < 3:
        return None

    d_arr = np.array(list(ld.values()))
    tol   = d_arr.mean() + 2 * d_arr.std()
    d_arr = d_arr[d_arr < tol]
    if len(d_arr) < 3:
        return None

    max_dst = d_arr.max()
    rx, ry, rz = x.copy(), y.copy(), z.copy()
    mkt_p = None
    final_vox = None

    for ratio in np.arange(2.0, 5.0 + step, step):
        vox = ratio * max_dst
        # Voxel decimation: keep one point per vox×vox×vox cell
        key = (np.floor(x / vox).astype(np.int64) * 1_000_000_007 +
               np.floor(y / vox).astype(np.int64) * 7001 +
               np.floor(z / vox).astype(np.int64))
        _, idx = np.unique(key, return_index=True)
        rx, ry, rz = x[idx], y[idx], z[idx]

        ld2 = avg_point_dist(rx, ry, rz, thickness)
        if len(ld2) < 3:
            break
        d2 = np.array(list(ld2.values()))
        d2 = d2[~np.isnan(d2)]
        if len(d2) < 3:
            break

        # Mann-Kendall trend test (scipy kendalltau ≈ mk.test)
        # Use pymannkendall for a closer match to R's trend::mk.test
        res_mk = mk.original_test(d2)
        mkt_p  = res_mk.p
        final_vox = vox
        if mkt_p > alpha:
            break

    if mkt_p is None:
        return None
    return {"x": rx, "y": ry, "z": rz, "vox": final_vox, "p": mkt_p}

# ── 3. KDE Shannon entropy in XY, XZ, YZ planes ───────────────────
def canopy_entropy(x, y, z, bw=0.2, grid_size=0.1):
    """
    KDE Shannon entropy in three orthogonal planes (Liu et al. 2022).
    Note: scipy gaussian_kde uses a bandwidth *factor* of data std;
    to approximate the absolute bw=0.2 m used by ks::kde in R, we
    scale by the per-plane std. Results are numerically close but not
    identical to the R implementation.
    """
    planes = {"XY": (x, y), "XZ": (x, z), "YZ": (y, z)}
    ce = {}
    for name, (a, b) in planes.items():
        # Approximate absolute bandwidth: bw_factor = bw / data_std
        std_a = a.std() or 1.0
        std_b = b.std() or 1.0
        bw_f  = bw / ((std_a + std_b) / 2)
        kde   = gaussian_kde(np.vstack([a, b]), bw_method=bw_f)

        g_a = np.arange(a.min() - 4*bw, a.max() + 4*bw, grid_size)
        g_b = np.arange(b.min() - 4*bw, b.max() + 4*bw, grid_size)
        ga, gb = np.meshgrid(g_a, g_b)
        density = kde(np.vstack([ga.ravel(), gb.ravel()]))
        density = density[density > 0]
        ce[name] = -np.sum(density * np.log(density) * grid_size**2)

    ce["XYZ"] = np.sqrt(ce["XY"]**2 + ce["XZ"]**2 + ce["YZ"]**2)
    return ce

# ── 4. Entropy wrapper with adaptive resampling ────────────────────
def get_entropy(x, y, h, thickness=1.0):
    """Use height above ground (h) for all geometry (Liu et al. 2022)."""
    res = adaptative_resample(x, y, h, thickness)
    if res is None:
        return None
    ent = canopy_entropy(res["x"], res["y"], res["z"])
    ent["p"]   = res["p"]
    ent["vox"] = res["vox"]
    return ent

# ── 5. Pixel-level complexity ──────────────────────────────────────
def pix_complexity(x_cell, y_cell, z_cell, h_cell):
    """Pixel-level wrapper — mirrors lidR pix_complexity."""
    keep = h_cell >= 1.0
    if keep.sum() < 10:
        return None
    try:
        return get_entropy(x_cell[keep], y_cell[keep], h_cell[keep])
    except Exception as e:
        print(f"get_entropy failed: {e}")
        return None

# Apply over 20 m grid using the pixel_metrics() helper from Section 1.1
def pixel_entropy(x, y, H, res=20):
    xmin, xmax = x.min(), x.max()
    ymin, ymax = y.min(), y.max()
    cols = max(1, int(np.ceil((xmax - xmin) / res)))
    rows = max(1, int(np.ceil((ymax - ymin) / res)))
    ci = np.clip(((x - xmin) / res).astype(int), 0, cols - 1)
    ri = np.clip(((ymax - y) / res).astype(int), 0, rows - 1)
    out = {}
    for r in range(rows):
        for c in range(cols):
            mask = (ri == r) & (ci == c)
            if mask.sum() < 10:
                continue
            m = pix_complexity(x[mask], y[mask], z[mask], H[mask])
            if m is None:
                continue
            for k, v in m.items():
                if k not in out:
                    out[k] = np.full((rows, cols), np.nan)
                out[k][r, c] = v
    return out

def roof_metrics(x, y, h):
    if len(h) < 6:
        return None
    pts    = np.column_stack([x, y, h])
    # PCA: eigenvector of smallest eigenvalue = surface normal
    _, evecs = np.linalg.eigh(np.cov(pts.T))
    normal   = evecs[:, 0]
    slope    = np.degrees(np.arccos(np.abs(normal[2])))
    return {
        "roof_slope_mean": slope,
        "roof_roughness":  np.std(h - h.mean())
    }

building_footprint_metrics <- function(x, y, z) {
  pts2d          <- st_sfc(st_multipoint(cbind(x, y)))
  hull           <- st_convex_hull(pts2d)
  footprint_area <- as.numeric(st_area(hull))
  perimeter      <- as.numeric(
    st_length(st_cast(hull, "MULTILINESTRING")))
  height_p95     <- quantile(z, 0.95)
  volume         <- footprint_area * height_p95
  compactness    <- 4 * pi * footprint_area / perimeter^2
  list(
    footprint_m2 = footprint_area,
    perimeter_m  = perimeter,
    compactness  = compactness,
    volume_m3    = volume,
    FAR          = volume / footprint_area
  )
}

# Approximate SVF via horizon scanning (simplified)
def svf_approx(ndsm, radius_px=50, n_dirs=36):
    rows, cols = ndsm.shape
    svf_out    = np.full_like(ndsm, np.nan, dtype=float)
    angles     = np.linspace(0, 2 * np.pi, n_dirs, endpoint=False)
    r_range    = np.arange(1, radius_px + 1)
    for i in range(rows):
        for j in range(cols):
            max_sin2 = 0.0
            for a in angles:
                ri = (i + r_range * np.sin(a)).astype(int)
                ci = (j + r_range * np.cos(a)).astype(int)
                valid = (ri >= 0) & (ri < rows) & (ci >= 0) & (ci < cols)
                if valid.any():
                    elev = ndsm[ri[valid], ci[valid]] / r_range[valid]
                    max_sin2 += np.sin(np.arctan(elev.max()))**2
            svf_out[i, j] = 1 - max_sin2 / n_dirs
    return svf_out
